    /** Tanggal kas dalam zona waktu aplikasi (kolom created_at bertipe timestamptz). */
    private function tglKas(): string
    {
        return '(created_at at time zone ?::text)::date';
    }

    private function rentang(string $p, Carbon $today): array
    {
        if ($p === 'hari') {
            return [$today->copy(), $today->copy(), $today->copy()->subDay(), $today->copy()->subDay()];
        }
        if ($p === 'minggu') {
            $f = $today->copy()->subDays(6);
            return [$f, $today->copy(), $f->copy()->subDays(7), $f->copy()->subDay()];
        }
        if ($p === 'bulan_lalu') {
            $f  = $today->copy()->subMonthNoOverflow()->startOfMonth();
            $pf = $f->copy()->subMonthNoOverflow()->startOfMonth();
            return [$f, $f->copy()->endOfMonth()->startOfDay(), $pf, $pf->copy()->endOfMonth()->startOfDay()];
        }
        $f  = $today->copy()->startOfMonth();
        $pf = $f->copy()->subMonthNoOverflow();
        $pt = $pf->copy()->addDays($today->day - 1);
        if ($pt->gt($pf->copy()->endOfMonth())) {
            $pt = $pf->copy()->endOfMonth();
        }
        return [$f, $today->copy(), $pf, $pt->startOfDay()];
    }

    private function ringkas(Carbon $from, Carbon $to, ?string $kasirId): array
    {
        $f = $from->toDateString();
        $t = $to->toDateString();

        $s = DB::table('sales')->where('is_voided', false)->whereBetween('sale_date', [$f, $t])
            ->when($kasirId, fn ($q) => $q->where('cashier_id', $kasirId))
            ->selectRaw('coalesce(sum(total_amount),0) as omzet, count(*) as trx')->first();

        $kg = DB::table('sale_items as i')->join('sales as s', 's.id', '=', 'i.sale_id')
            ->where('s.is_voided', false)->whereBetween('s.sale_date', [$f, $t])
            ->whereRaw("upper(i.unit) = 'KG'")
            ->when($kasirId, fn ($q) => $q->where('s.cashier_id', $kasirId))
            ->sum('i.quantity');

        $kas = DB::table('payments')->whereNull('voided_at')
            ->whereRaw($this->tglKas() . ' between ? and ?', [config('app.timezone'), $f, $t])
            ->when($kasirId, fn ($q) => $q->where('cashier_id', $kasirId))
            ->sum('amount');

        return ['omzet' => (float) $s->omzet, 'trx' => (int) $s->trx, 'kg' => (float) $kg, 'kas' => (float) $kas];
    }

                'status'       => 'VOIDED',
                'void_reason'  => $data['void_reason'],
                'voided_at'    => now(),
                'voided_by_id' => Auth::id(),

    public function store(Request $request)
    {
        $this->wajibBolehTulis();

        $data = $request->validate([
            'supplier_id'       => ['bail', 'required', 'uuid', 'exists:suppliers,id'],
            'purchase_date'     => ['required', 'date', 'before_or_equal:today'],
            'total_birds'       => ['required', 'integer', 'min:1', 'max:1000000'],
            'total_weight_kg'   => ['required', 'numeric', 'min:0.01', 'max:100000'],
            'unit_price_per_kg' => ['required', 'numeric', 'min:1', 'max:9999999'],
            'notes'             => ['nullable', 'string', 'max:255'],
        ]);

        $supplier = Supplier::where('id', $data['supplier_id'])->where('is_active', true)->first();

        if (! $supplier) {
            throw ValidationException::withMessages([
                'supplier_id' => 'Pemasok ini sudah tidak aktif. Pilih pemasok lain.',
            ]);
        }

        $purchase = DB::transaction(function () use ($data, $supplier) {
            $tanggal = Carbon::parse($data['purchase_date']);

            // Nomor transaksi: PB-TanggalPembelian-Urut
            $this->kunci('purchase_number_' . $tanggal->format('Ymd'));

            $prefix   = 'PB-' . $tanggal->format('Ymd') . '-';
            $terakhir = PurchaseTransaction::where('transaction_number', 'like', $prefix . '%')->max('transaction_number');
            $urut     = $terakhir ? ((int) substr($terakhir, strlen($prefix))) + 1 : 1;

            // Total biaya selalu dihitung di server, bukan dari isian browser
            $berat = round((float) $data['total_weight_kg'], 2);
            $harga = round((float) $data['unit_price_per_kg'], 2);

            $purchase = PurchaseTransaction::create([
                'transaction_number' => $prefix . str_pad((string) $urut, 4, '0', STR_PAD_LEFT),
                'purchase_date'      => $tanggal->toDateString(),
                'supplier_id'        => $supplier->id,
                'total_birds'        => (int) $data['total_birds'],
                'total_weight_kg'    => $berat,
                'unit_price_per_kg'  => $harga,
                'total_cost'         => round($berat * $harga, 2),
                'notes'              => $data['notes'] ?? null,
                'received_by_id'     => Auth::id(),
            ]);

            AuditLog::create([
                'action'     => 'CREATE',
                'table_name' => 'purchase_transactions',
                'record_id'  => $purchase->id,
                'new_values' => $purchase->getAttributes(),
                'user_id'    => Auth::id(),
            ]);

            return $purchase;
        });

        return redirect()->route('pembelian.show', $purchase)
            ->with('status', 'Pembelian tersimpan: ' . $purchase->transaction_number . '.');
    }

    public function show(PurchaseTransaction $purchase)
    {
        $purchase->load(['supplier', 'receivedBy']);

        return view('pembelian.show', compact('purchase'));
    }
