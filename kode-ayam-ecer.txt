<?php

namespace App\Http\Controllers;

use App\Models\AuditLog;
use App\Models\LiveBirdSale;
use App\Services\StokService;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Auth;
use Illuminate\Support\Facades\DB;
use Illuminate\Validation\ValidationException;

class AyamJualController extends Controller
{
    // Penjualan ayam hidup yang dicatat Pengawas Kandang (ecer). Harga diisi kasir nanti.
    public function store(Request $request, StokService $stok)
    {
        abort_unless(Auth::user()->canWrite('kandang'), 403);

        $data = $request->validate([
            'sale_date'     => \App\Support\Pencatatan::aturan(),
            'customer_name' => ['required', 'string', 'max:100'],
            'birds'         => ['required', 'integer', 'min:1', 'max:100000'],
            'weight_kg'     => ['required', 'numeric', 'min:0.01', 'max:100000'],
            'sale_notes'    => ['nullable', 'string', 'max:255'],
        ]);

        DB::transaction(function () use ($data, $stok) {
            $stok->kunci('stok_ayam_hidup');
            $sisa = $stok->ayamHidup();

            if ($data['birds'] > $sisa['ekor']) {
                throw ValidationException::withMessages([
                    'birds' => 'Jumlah melebihi stok ayam hidup saat ini (' . number_format($sisa['ekor'], 0, ',', '.') . ' ekor).',
                ]);
            }

            $rec = LiveBirdSale::create([
                'sale_date'     => $data['sale_date'],
                'customer_name' => trim($data['customer_name']),
                'birds'         => (int) $data['birds'],
                'weight_kg'     => round((float) $data['weight_kg'], 2),
                'notes'         => $data['sale_notes'] ?? null,
                'created_by_id' => Auth::id(),
            ]);

            AuditLog::create([
                'action'     => 'CREATE',
                'table_name' => 'live_bird_sales',
                'record_id'  => $rec->id,
                'new_values' => $rec->getAttributes(),
                'user_id'    => Auth::id(),
            ]);
        });

        return redirect()->route('ayamkeluar.index')->with('status', 'Penjualan ayam hidup tersimpan.');
    }
}

<?php

namespace App\Http\Controllers;

use App\Models\AuditLog;
use App\Models\ProductionBatch;
use App\Services\StokService;
use Illuminate\Http\Request;
use Illuminate\Support\Carbon;
use Illuminate\Support\Facades\Auth;
use Illuminate\Support\Facades\DB;
use Illuminate\Validation\ValidationException;

class AyamKeluarController extends Controller
{
    public function index(StokService $stok)
    {
        // Riwayat produksi dan penjualan ada di halaman Riwayat Ayam (RiwayatAyamController).
        $ecerBulan = DB::table('live_bird_sales')
            ->whereBetween('sale_date', [now()->startOfMonth()->toDateString(), now()->toDateString()])
            ->selectRaw('coalesce(sum(birds),0) as e, coalesce(sum(weight_kg),0) as k')->first();

        return view('ayam-keluar.index', [
            'stok'       => $stok->ayamHidup(),
            'ecerBulan'  => $ecerBulan,
            'bolehTulis' => Auth::user()->canWrite('kandang'),
        ]);
    }

    public function store(Request $request, StokService $stok)
    {
        abort_unless(Auth::user()->canWrite('kandang'), 403);

        $data = $request->validate([
            'production_date' => \App\Support\Pencatatan::aturan(),
            'input_birds'     => ['required', 'integer', 'min:1', 'max:1000000'],
            'input_weight_kg' => ['required', 'numeric', 'min:0.01', 'max:1000000'],
            'notes'           => ['nullable', 'string', 'max:255'],
        ]);

        $batch = DB::transaction(function () use ($data, $stok) {
            $tanggal = Carbon::parse($data['production_date']);
            $ymd = $tanggal->format('Ymd');

            $stok->kunci('stok_ayam_hidup');
            $stok->kunci('produksi_number_' . $ymd);

            $sisa = $stok->ayamHidup();
            if ($data['input_birds'] > $sisa['ekor']) {
                throw ValidationException::withMessages([
                    'input_birds' => 'Jumlah ayam melebihi stok ayam hidup saat ini (' . number_format($sisa['ekor'], 0, ',', '.') . ' ekor).',
                ]);
            }

            $kg = round((float) $data['input_weight_kg'], 2);

            // Satu batch per tanggal: pakai yang sudah ada (dibuat Kasir atau Pengawas), kalau belum ada buat baru
            $batch = ProductionBatch::where('production_date', $tanggal->toDateString())
                ->orderBy('created_at')->lockForUpdate()->first();

            if ($batch) {
                $batch->update([
                    'input_birds'     => (int) $batch->input_birds + (int) $data['input_birds'],
                    'input_weight_kg' => round((float) $batch->input_weight_kg + $kg, 2),
                    'status'          => 'COMPLETED',
                    'notes'           => $batch->notes ?: ($data['notes'] ?? null),
                ]);
            } else {
                $prefix = 'PRD-' . $ymd . '-';
                $terakhir = ProductionBatch::where('batch_number', 'like', $prefix . '%')->max('batch_number');
                $urut = $terakhir ? ((int) substr($terakhir, strlen($prefix))) + 1 : 1;

                $batch = ProductionBatch::create([
                    'batch_number'    => $prefix . str_pad((string) $urut, 4, '0', STR_PAD_LEFT),
                    'production_date' => $tanggal->toDateString(),
                    'input_birds'     => (int) $data['input_birds'],
                    'input_weight_kg' => $kg,
                    'status'          => 'COMPLETED',
                    'notes'           => $data['notes'] ?? null,
                    'created_by_id'   => Auth::id(),
                ]);
            }

            AuditLog::create([
                'action'     => 'CREATE',
                'table_name' => 'production_batches',
                'record_id'  => $batch->id,
                'new_values' => ['batch_number' => $batch->batch_number, 'tambah_ekor' => (int) $data['input_birds'], 'tambah_kg' => $kg],
                'user_id'    => Auth::id(),
            ]);

            return $batch->fresh();
        });

        return redirect()->route('ayamkeluar.index')
            ->with('status', 'Ayam masuk produksi tersimpan (' . $batch->batch_number . '). Total tanggal itu: ' . number_format($batch->input_birds, 0, ',', '.') . ' ekor.');
    }
}

<?php

namespace App\Http\Controllers;

use Illuminate\Http\Request;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\Hash;
use Illuminate\Support\Facades\Schema;
use Illuminate\Support\Str;
use Illuminate\Validation\ValidationException;

class HapusDataController extends Controller
{
    /** Tabel yang kolom referensinya hanya dikosongkan (bukan ikut dihapus) saat induknya dihapus. */
    private const LEPAS = ['live_bird_sales'];

    /**
     * Kelompok DATA TRANSAKSI yang boleh dihapus (BUKAN data induk).
     * Tabel yang tidak ada di database otomatis dilewati.
     * Ingin menambah/mengurangi? Edit daftar ini saja.
     */
    private function kelompok(): array
    {
        return [
            'KASIR' => [
                'judul' => 'Data Input Kasir',
                'tabel' => [
                    'sales'      => 'Penjualan (Nota)',
                    'sale_items' => 'Item Penjualan',
                    'payments'   => 'Pembayaran',
                ],
            ],
            'PENGAWAS' => [
                'judul' => 'Data Input Pengawas Kandang',
                'tabel' => [
                    'purchase_transactions' => 'Pembelian',
                    'production_batches'    => 'Batch Produksi',
                    'weighing_records'      => 'Penimbangan',
                    'live_bird_sales'       => 'Penjualan Ayam Hidup (Ecer)',
                    'live_bird_mortalities' => 'Ayam Mati',
                    'live_bird_opnames'     => 'Opname Ayam Hidup',
                    'kitchen_orders'        => 'Pesanan Dapur',
                    'kitchen_order_lines'   => 'Rincian Pesanan Dapur',
                    'stock_movements'       => 'Mutasi Stok',
                    'stock_adjustments'     => 'Penyesuaian Stok',
                ],
            ],
        ];
    }

    private function semuaTabel(): array
    {
        $hasil = [];
        foreach ($this->kelompok() as $k) {
            $hasil += $k['tabel'];
        }
        return $hasil;
    }

    private function izin(Request $r): void
    {
        // Hanya admin. Ganti bila pemilik/manager juga boleh, misalnya:
        // abort_unless($r->user()->isAdmin() || $r->user()->isManager(), 403);
        abort_unless($r->user() && $r->user()->isAdmin(), 403);
    }

    private function cekPassword(Request $r): void
    {
        $r->validate(['password' => 'required']);
        if (!Hash::check($r->password, $r->user()->password)) {
            throw ValidationException::withMessages(['password' => 'Password salah.']);
        }
    }

    private function pastikanTabel(string $tabel): void
    {
        abort_unless(
            array_key_exists($tabel, $this->semuaTabel())
            && Schema::hasTable($tabel)
            && Schema::hasColumn($tabel, 'id'),
            404
        );
    }

    /**
     * Hapus baris + semua data turunannya (tabel anak dengan kolom <induk>_id),
     * supaya tidak ada data yatim.
     */
    private function hapusBaris(string $tabel, array $ids, int $depth = 0): void
    {
        if (!$ids || $depth > 5 || !Schema::hasColumn($tabel, 'id')) {
            return;
        }

        $kolom = Str::singular($tabel) . '_id';
        foreach (array_keys($this->semuaTabel()) as $anak) {
            if ($anak === $tabel || !Schema::hasTable($anak)
                || !Schema::hasColumn($anak, $kolom) || !Schema::hasColumn($anak, 'id')) {
                continue;
            }
            if (in_array($anak, self::LEPAS, true)) {
                DB::table($anak)->whereIn($kolom, $ids)->update([$kolom => null]);
                continue;
            }
            $idAnak = DB::table($anak)->whereIn($kolom, $ids)->pluck('id')->all();
            $this->hapusBaris($anak, $idAnak, $depth + 1);
        }

        DB::table($tabel)->whereIn('id', $ids)->delete();
    }

    public function index(Request $r)
    {
        $this->izin($r);

        $kelompok = $this->kelompok();
        foreach ($this->kelompok() as $kode => $k) {
            foreach ($k['tabel'] as $t => $label) {
                if (!Schema::hasTable($t)) {
                    unset($kelompok[$kode]['tabel'][$t]);
                    continue;
                }
                $kelompok[$kode]['tabel'][$t] = ['label' => $label, 'jumlah' => DB::table($t)->count()];
            }
        }

        return view('hapus-data.index', compact('kelompok'));
    }

    public function show(Request $r, string $tabel)
    {
        $this->izin($r);
        $this->pastikanTabel($tabel);

        $label = $this->semuaTabel()[$tabel];
        $urut = Schema::hasColumn($tabel, 'created_at') ? 'created_at' : 'id';
        $baris = DB::table($tabel)->orderByDesc($urut)->simplePaginate(50);

        return view('hapus-data.show', compact('tabel', 'label', 'baris'));
    }

    /** Hapus PERMANEN seluruh isi tabel yang dicentang. */
    public function destroyTables(Request $r)
    {
        $this->izin($r);
        $r->validate([
            'tabel'      => 'required|array|min:1',
            'tabel.*'    => 'string',
            'konfirmasi' => 'required|in:HAPUS',
        ], [
            'tabel.required' => 'Pilih minimal satu data.',
            'konfirmasi.in'  => 'Ketik HAPUS (huruf besar) untuk konfirmasi.',
        ]);
        $this->cekPassword($r);

        foreach ($r->tabel as $t) {
            $this->pastikanTabel($t);
        }

        Schema::disableForeignKeyConstraints();
        try {
            DB::transaction(function () use ($r) {
                foreach (array_keys($this->semuaTabel()) as $t) {
                    if (!in_array($t, $r->tabel, true) || !Schema::hasTable($t)) {
                        continue;
                    }
                    DB::table($t)->pluck('id')->chunk(500)->each(function ($ids) use ($t) {
                        $this->hapusBaris($t, $ids->all());
                    });
                }
            });
        } finally {
            Schema::enableForeignKeyConstraints();
        }

        return redirect()->route('hapus-data.index')->with('sukses', 'Data terpilih berhasil dihapus permanen.');
    }

    /** Hapus PERMANEN baris yang dicentang pada satu tabel. */
    public function destroyRows(Request $r, string $tabel)
    {
        $this->izin($r);
        $this->pastikanTabel($tabel);

        $r->validate([
            'ids'   => 'required|array|min:1',
            'ids.*' => 'string',
        ], ['ids.required' => 'Pilih minimal satu baris.']);
        $this->cekPassword($r);

        Schema::disableForeignKeyConstraints();
        try {
            DB::transaction(function () use ($r, $tabel) {
                $this->hapusBaris($tabel, array_values($r->ids));
            });
        } finally {
            Schema::enableForeignKeyConstraints();
        }

        return redirect()->route('hapus-data.show', $tabel)
            ->with('sukses', count($r->ids) . ' baris berhasil dihapus permanen.');
    }
}

===== app\Http\Controllers\KasirAyamHidupController.php =====
<?php

namespace App\Http\Controllers;

use App\Models\AuditLog;
use App\Models\Customer;
use App\Models\CustomerType;
use App\Models\LiveBirdSale;
use App\Models\Payment;
use App\Models\Product;
use App\Models\ProductPrice;
use App\Models\Sale;
use App\Services\PiutangService;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Auth;
use Illuminate\Support\Facades\DB;
use Illuminate\Validation\ValidationException;

/**
 * Nota kasir untuk ayam hidup ecer yang sudah dicatat pengawas.
 * Jumlah ekor dan berat diambil dari catatan pengawas. Stok tidak dicek lagi di sini
 * karena sudah berkurang saat pengawas mencatat, dan tidak boleh terhitung dua kali.
 */
class KasirAyamHidupController extends Controller
{
    private function kunci(string $kunci): void
    {
        DB::select('select pg_advisory_xact_lock(?)', [(int) crc32($kunci)]);
    }

    /** Catatan pengawas tanpa nota aktif (belum dinota, atau notanya dibatalkan). */
    private function belumDinota()
    {
        return LiveBirdSale::query()
            ->leftJoin('sales as s', 's.id', '=', 'live_bird_sales.sale_id')
            ->where(fn ($q) => $q->whereNull('live_bird_sales.sale_id')->orWhere(fn ($v) => $v->where('s.is_voided', true)->where(fn ($z) => $z->whereNull('live_bird_sales.sumber')->orWhere('live_bird_sales.sumber', '!=', 'KASIR'))))
            ->select('live_bird_sales.*');
    }

    private function produk(): Product
    {
        $p = Product::where('is_live_bird', true)->where('is_active', true)->first();
        abort_if(! $p, 409, 'Produk Ayam Hidup belum dibuat atau tidak aktif. Hubungi admin.');

        return $p;
    }

    public function index()
    {
        abort_unless(Auth::user()->canView('kasir'), 403);

        $daftar = $this->belumDinota()
            ->orderBy('live_bird_sales.sale_date')->orderBy('live_bird_sales.created_at')
            ->get();

        $sudah = DB::table('live_bird_sales as l')
            ->join('sales as s', 's.id', '=', 'l.sale_id')
            ->where('s.is_voided', false)
            ->orderByDesc('l.sale_date')->orderByDesc('l.created_at')
            ->limit(20)
            ->get(['l.sale_date', 'l.customer_name', 'l.birds', 'l.weight_kg', 's.id as sale_id', 's.sale_number']);

        return view('kasir.ayam-hidup.index', compact('daftar', 'sudah'));
    }

    public function create(string $rec, PiutangService $piutang)
    {
        abort_unless(Auth::user()->canWrite('kasir'), 403);

        $catatan = $this->belumDinota()->where('live_bird_sales.id', $rec)->firstOrFail();
        $produk  = $this->produk();

        $types = CustomerType::where('is_active', true)->orderBy('name')->get(['id', 'code', 'name']);
        $defaultTypeId = $types->firstWhere('code', 'MASYARAKAT_UMUM')?->id ?? $types->first()?->id;

        $harga = ProductPrice::where('product_id', $produk->id)->where('is_active', true)
            ->pluck('price', 'customer_type_id');

        $nama = mb_strtolower(trim((string) $catatan->customer_name));
        $pelanggan = $nama !== '' ? Customer::whereRaw('lower(name) = ?', [$nama])->first() : null;
        $saldo = $pelanggan ? $piutang->saldo($pelanggan->id) : 0.0;
        $defaultTempo = (int) config('kasir.tempo_hari', 7);

        return view('kasir.ayam-hidup.create', compact('catatan', 'types', 'defaultTypeId', 'harga', 'saldo', 'defaultTempo'));
    }

    public function store(Request $request, PiutangService $piutang, string $rec)
    {
        abort_unless(Auth::user()->canWrite('kasir'), 403);

        $data = $request->validate([
            'customer_type_id' => ['required', 'exists:customer_types,id'],
            'sale_date'        => \App\Support\Pencatatan::aturan(false),
            'discount'         => ['nullable', 'numeric', 'min:0'],
            'payment_method'   => ['required', 'in:CASH,TRANSFER,HUTANG'],
            'amount_paid'      => ['nullable', 'numeric', 'min:0'],
            'reference_number' => ['nullable', 'string', 'max:100'],
            'due_days'         => ['nullable', 'integer', 'min:0', 'max:365'],
        ]);

        [$sale, $kembalian, $sisaHutang] = DB::transaction(function () use ($rec, $data, $piutang) {
            $produk  = $this->produk();
            $tglJual = \App\Support\Pencatatan::tanggal($data['sale_date'] ?? null);
            $today   = $tglJual->toDateString();

            // Urutan kunci: nomor nota -> catatan pengawas -> nama pelanggan -> id pelanggan
            $this->kunci('sale_number_' . $today);

            $catatan = LiveBirdSale::whereKey($rec)->lockForUpdate()->firstOrFail();

            if ($catatan->sale_id && ! Sale::whereKey($catatan->sale_id)->value('is_voided')) {
                throw ValidationException::withMessages(['payment_method' => 'Catatan ini sudah dibuatkan nota.']);
            }

            $harga = (float) ProductPrice::where('product_id', $produk->id)
                ->where('customer_type_id', $data['customer_type_id'])
                ->where('is_active', true)->value('price');

            if ($harga <= 0) {
                throw ValidationException::withMessages([
                    'customer_type_id' => 'Harga Ayam Hidup untuk jenis pelanggan ini belum diatur.',
                ]);
            }

            $ekor     = (int) $catatan->birds;
            $berat    = round((float) $catatan->weight_kg, 2);
            $subtotal = round($harga * $ekor, 2);
            $discount = round((float) ($data['discount'] ?? 0), 2);

            if ($discount > $subtotal) {
                throw ValidationException::withMessages(['discount' => 'Diskon tidak boleh melebihi subtotal.']);
            }

            $total  = round($subtotal - $discount, 2);
            $metode = $data['payment_method'];
            $paid   = round((float) ($data['amount_paid'] ?? 0), 2);
            $tempo  = (int) ($data['due_days'] ?? config('kasir.tempo_hari', 7));

            $nama    = trim((string) $catatan->customer_name);
            $adaNama = $nama !== '' && mb_strtolower($nama) !== 'umum';

            $customer = null;
            if ($adaNama) {
                $this->kunci('cust_name_' . mb_strtolower($nama));

                $customer = Customer::whereRaw('lower(name) = ?', [mb_strtolower($nama)])->first()
                    ?? Customer::create([
                        'name'             => $nama,
                        'customer_type_id' => $data['customer_type_id'],
                        'is_active'        => true,
                    ]);

                $this->kunci('cust_' . $customer->id);
            }

            if ($metode === 'HUTANG' && ! $customer) {
                throw ValidationException::withMessages([
                    'payment_method' => 'Hutang harus atas nama pelanggan, sedangkan catatan ini atas nama "Umum".',
                ]);
            }

            if ($metode !== 'HUTANG' && $paid < $total) {
                throw ValidationException::withMessages([
                    'amount_paid' => 'Uang dibayar kurang dari total. Pilih "Hutang" kalau pelanggan berhutang.',
                ]);
            }

            $hutangLama = $customer ? $piutang->saldo($customer->id) : 0.0;
            $tagihan    = round($total + $hutangLama, 2);

            if ($metode === 'TRANSFER' && $paid > $tagihan + 0.004) {
                throw ValidationException::withMessages([
                    'amount_paid' => 'Transfer tidak boleh melebihi total tagihan (Rp ' . number_format($tagihan, 0, ',', '.') . ').',
                ]);
            }

            $dibayar = round(min($paid, $tagihan), 2);

            $prefix   = 'INV-' . $tglJual->format('Ymd') . '-';
            $terakhir = Sale::where('sale_number', 'like', $prefix . '%')->max('sale_number');
            $urut     = $terakhir ? ((int) substr($terakhir, strlen($prefix))) + 1 : 1;
            $number   = $prefix . str_pad((string) $urut, 4, '0', STR_PAD_LEFT);

            $sale = Sale::create([
                'sale_number'      => $number,
                'sale_date'        => $today,
                'due_date'         => $customer ? $tglJual->copy()->addDays($tempo)->toDateString() : null,
                'customer_id'      => $customer?->id,
                'customer_name'    => $customer?->name ?? 'Umum',
                'customer_type_id' => $data['customer_type_id'],
                'subtotal'         => $subtotal,
                'discount'         => $discount,
                'total_amount'     => $total,
                'previous_balance' => $hutangLama,
                'balance_after'    => 0,
                'status'           => 'COMPLETED',
                'cashier_id'       => Auth::id(),
            ]);

            $sale->items()->create([
                'product_id' => $produk->id,
                'quantity'   => $ekor,
                'unit'       => $produk->primary_unit ?? 'EKOR',
                'unit_price' => $harga,
                'subtotal'   => $subtotal,
                'weight_kg'  => $berat,
            ]);

            $metodeBayar = $metode === 'HUTANG' ? 'CASH' : $metode;
            $teralokasi  = 0.0;

            if ($dibayar > 0) {
                if ($customer) {
                    $teralokasi = $piutang->bayar(
                        $customer->id,
                        $dibayar,
                        $metodeBayar,
                        $data['reference_number'] ?? null,
                        Auth::id(), $sale->id, $tglJual);
                } else {
                    Payment::create([
                        'sale_id'            => $sale->id,
                        'payment_method'     => $metodeBayar,
                        'amount'             => $dibayar,
                        'reference_number'   => $data['reference_number'] ?? null,
                        'cashier_id'         => Auth::id(),
                        'settlement_sale_id' => $sale->id,
                        'created_at'         => \App\Support\Pencatatan::waktu($tglJual),
                    ]);
                    $teralokasi = $dibayar;
                }
            }

            $kembalian  = round($paid - $teralokasi, 2);
            $sisaHutang = $customer ? $piutang->saldo($customer->id) : 0.0;

            $tertua = ($customer && $sisaHutang > 0.004)
                ? $piutang->notaTerbuka($customer->id)->min('jatuh_tempo')
                : null;

            $sale->update([
                'balance_after'    => $sisaHutang,
                'balance_due_date' => $tertua,
            ]);

            // Sambungkan catatan pengawas ke nota ini: stok kini dihitung dari nota, bukan dari catatan
            $catatan->update(['sale_id' => $sale->id]);

            AuditLog::create([
                'action'     => 'UPDATE',
                'table_name' => 'live_bird_sales',
                'record_id'  => $catatan->id,
                'new_values' => ['sale_id' => $sale->id, 'sale_number' => $sale->sale_number],
                'user_id'    => Auth::id(),
            ]);

    public function store(Request $request, PiutangService $piutang, StokService $stok)
    {
        abort_unless(Auth::user()->canWrite('kasir'), 403);

            'birds'            => ['required', 'integer', 'min:1', 'max:99999'],

            'amount_paid'      => ['nullable', 'numeric', 'min:0'],
            'reference_number' => ['nullable', 'string', 'max:100'],
            'due_days'         => ['nullable', 'integer', 'min:0', 'max:365'],

            $produk  = $this->produk();
            $tglJual = \App\Support\Pencatatan::tanggal($data['sale_date'] ?? null);
            $today   = $tglJual->toDateString();

            // Urutan kunci (sama dengan KasirController): nomor nota -> nama pelanggan -> id pelanggan -> stok ayam hidup
            $this->kunci('sale_number_' . $today);

                        'is_active'        => true,

            $berat = round((float) $data['weight_kg'], 2);

            $stok->kunci('stok_ayam_hidup');

            }

            $metode = $data['payment_method'];
            $paid   = round((float) ($data['amount_paid'] ?? 0), 2);
            $tempo  = (int) ($data['due_days'] ?? config('kasir.tempo_hari', 7));

                    'payment_method' => 'Hutang harus atas nama pelanggan. Isi nama pelanggan (bukan "Umum").',

            if ($metode !== 'HUTANG' && $paid < $total) {
                throw ValidationException::withMessages([
                    'amount_paid' => 'Uang dibayar kurang dari total. Pilih "Hutang" kalau pelanggan berhutang.',
                ]);
            }

            $hutangLama = $customer ? $piutang->saldo($customer->id) : 0.0;
            $tagihan    = round($total + $hutangLama, 2);

                'unit'       => $produk->primary_unit ?? 'EKOR',
                'unit_price' => $harga,
                'subtotal'   => $subtotal,
                'weight_kg'  => $berat,

                        'settlement_sale_id' => $sale->id,
                        'created_at'         => \App\Support\Pencatatan::waktu($tglJual),
                    ]);
                    $teralokasi = $dibayar;
                }
            }

            $kembalian  = round($paid - $teralokasi, 2);
            $sisaHutang = $customer ? $piutang->saldo($customer->id) : 0.0;

                : null;

            $sale->update([
                'balance_after'    => $sisaHutang,
                'balance_due_date' => $tertua,

                'created_by_id' => Auth::id(),

                'table_name' => 'live_bird_sales',
                'record_id'  => $rec->id,
                'new_values' => ['sale_id' => $sale->id, 'sale_number' => $sale->sale_number, 'birds' => $ekor, 'weight_kg' => $berat, 'sumber' => 'KASIR'],
                'user_id'    => Auth::id(),
            ]);

            \App\Support\Pencatatan::catatMundur('sales', (string) $sale->id, $tglJual, ['sale_number' => $sale->sale_number]);

            return [$sale, $kembalian, $sisaHutang];
        });

        $pesan = 'Penjualan ayam ecer tersimpan.';

                        $teralokasi = $dibayar;
                    }
                }

                // Kembalian dihitung dari uang yang benar-benar tercatat
                $kembalian  = round($paid - $teralokasi, 2);
                $sisaHutang = $customer ? $piutang->saldo($customer->id) : 0.0;

                // Jatuh tempo hutang tertua yang masih tersisa, untuk dicetak di nota
                $tertua = ($customer && $sisaHutang > 0.004)
                    ? $piutang->notaTerbuka($customer->id)->min('jatuh_tempo')
                    : null;

                $sale->update([
                    'balance_after'    => $sisaHutang,
                    'balance_due_date' => $tertua,
                ]);

                \App\Support\Pencatatan::catatMundur('sales', (string) $sale->id, $tglJual, ['sale_number' => $sale->sale_number]);

                return [$sale, $kembalian, $sisaHutang];
            }
        );

        $pesan = 'Transaksi tersimpan.';
        if ($kembalian > 0.004) {
            $pesan .= ' Kembalian: Rp ' . number_format($kembalian, 0, ',', '.') . '.';
        }
        if ($sisaHutang > 0.004) {
            $pesan .= ' Sisa hutang pelanggan: Rp ' . number_format($sisaHutang, 0, ',', '.') . '.';
        }

        return redirect()->route('kasir.show', $sale)->with('status', $pesan);
    }

        $dibayar = (float) $sale->settlementPayments

    public function void(Request $request, Sale $sale)
    {
        abort_unless(Auth::user()->isAdmin(), 403, 'Hanya admin yang boleh membatalkan transaksi.');

        $data = $request->validate([
            'void_reason' => ['required', 'string', 'min:5', 'max:255'],
        ]);

        DB::transaction(function () use ($sale, $data) {
            // Kunci per pelanggan, sama seperti saat transaksi dibuat
            if ($sale->customer_id) {
                $this->kunci('cust_' . $sale->customer_id);
            }

            $sale = Sale::whereKey($sale->id)->lockForUpdate()->firstOrFail();

            if ($sale->is_voided) {
                throw ValidationException::withMessages(['void_reason' => 'Transaksi ini sudah dibatalkan.']);
            }

            // Nota ini sudah dibayar lewat transaksi lain / halaman Piutang -> perlu penanganan manual
            $dibayarTerpisah = Payment::where('sale_id', $sale->id)
                ->whereNull('voided_at')
                ->where(function ($q) use ($sale) {
                    $q->whereNull('settlement_sale_id')
                      ->orWhere('settlement_sale_id', '!=', $sale->id);
                })
                ->exists();

            if ($dibayarTerpisah) {
                throw ValidationException::withMessages([
                    'void_reason' => 'Nota ini sudah dibayar lewat transaksi lain. Tidak bisa dibatalkan otomatis karena ada uang yang harus dikembalikan.',
                ]);
            }

            // Balik semua uang yang diterima di nota ini (termasuk yang melunasi nota lama)
            Payment::where('settlement_sale_id', $sale->id)
                ->whereNull('voided_at')
                ->update(['voided_at' => now()]);

            $sale->update([
                'is_voided'    => true,
                'status'       => 'VOIDED',
                'void_reason'  => $data['void_reason'],
                'voided_at'    => now(),
                'voided_by_id' => Auth::id(),
            ]);
        });

        return redirect()->route('kasir.show', $sale)->with('status', 'Transaksi dibatalkan.');
    }

                ]);

                'opname_date'   => $data['opname_date'],
                'system_birds'  => $sis['ekor'],
                'actual_birds'  => $aktualE,
                'diff_birds'    => $selisihE,
                'system_kg'     => $sis['kg'],
                'actual_kg'     => $aktualK,
                'diff_kg'       => $selisihK,

<?php

namespace App\Http\Controllers;

use App\Models\LiveBirdSale;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Auth;

/**
 * Pengawas memverifikasi penjualan ayam ecer yang dicatat kasir.
 * Kalau angkanya beda, selisih dicatat (angka pengawas + alasan). Nota dan stok TIDAK diubah otomatis:
 * selisih fisik diselesaikan lewat Opname Ayam, selisih uang oleh admin.
 */
class VerifikasiAyamController extends Controller
{
    public function store(Request $request, string $rec)
    {
        abort_unless(Auth::user()->canWrite('kandang'), 403);

        $d = $request->validate([
            'birds'     => ['required', 'integer', 'min:0', 'max:99999'],
            'weight_kg' => ['required', 'numeric', 'min:0', 'max:99999'],
            'catatan'   => ['nullable', 'string', 'max:255'],
        ]);

        $selisih = DB::transaction(function () use ($rec, $d) {
            $r = LiveBirdSale::whereKey($rec)->lockForUpdate()->firstOrFail();

            if ($r->sumber !== 'KASIR') {
                throw ValidationException::withMessages(['birds' => 'Hanya penjualan yang dicatat kasir yang diverifikasi di sini.']);
            }
            if ($r->diverifikasi_at) {
                throw ValidationException::withMessages(['birds' => 'Catatan ini sudah diverifikasi.']);
            }
            if ((string) $r->created_by_id === (string) Auth::id() && ! Auth::user()->isAdmin()) {
                throw ValidationException::withMessages(['birds' => 'Anda tidak boleh memverifikasi catatan yang Anda buat sendiri.']);
            }

            $ekor  = (int) $d['birds'];
            $berat = round((float) $d['weight_kg'], 2);
            $sama  = $ekor === (int) $r->birds && abs($berat - (float) $r->weight_kg) < 0.005;
            $alasan = trim((string) ($d['catatan'] ?? ''));

            if (! $sama && $alasan === '') {
                throw ValidationException::withMessages(['catatan' => 'Angka berbeda dari catatan kasir. Isi alasannya.']);
            }

            $r->diverifikasi_oleh = (string) Auth::id();
            $r->diverifikasi_at   = now();
            if (! $sama) {
                $r->ekor_pengawas       = $ekor;
                $r->berat_pengawas      = $berat;
                $r->catatan_verifikasi  = $alasan;
            }
            $r->save();

            AuditLog::create([
                'action'     => 'UPDATE',
                'table_name' => 'live_bird_sales',
                'record_id'  => $r->id,
                'new_values' => $sama
                    ? ['verifikasi' => 'sesuai']
                    : ['verifikasi' => 'selisih', 'kasir' => ['birds' => (int) $r->birds, 'kg' => (float) $r->weight_kg], 'pengawas' => ['birds' => $ekor, 'kg' => $berat], 'alasan' => $alasan],
                'user_id'    => Auth::id(),
            ]);

            return ! $sama;
        });

        return back()->with('status', $selisih
            ? 'Selisih dicatat. Stok dan nota tidak diubah otomatis: selesaikan lewat Opname Ayam atau minta admin memeriksa nota.'
            : 'Penjualan terverifikasi.');
    }

===== app\Services\StokService.php =====
<?php

namespace App\Services;

use App\Models\Product;
use Illuminate\Support\Carbon;
use Illuminate\Support\Collection;
use Illuminate\Support\Facades\DB;

/**
 * Semua stok dihitung dari catatan sumbernya, bukan disimpan sebagai angka terpisah.
 * Jadi void nota, koreksi, dan opname otomatis ikut terhitung.
 */
class StokService
{
    /** Kunci antar-transaksi PostgreSQL (lepas otomatis saat transaksi selesai). */
    public function kunci(string $kunci): void
    {
        DB::select('select pg_advisory_xact_lock(?)', [(int) crc32($kunci)]);
    }

    /** Ayam hidup terjual per ekor. Tanpa tanggal = seluruh waktu. */
    private function jualEkor(?string $dari = null, ?string $sampai = null): object
    {
        return DB::table('sale_items as i')
            ->join('sales as s', 's.id', '=', 'i.sale_id')
            ->join('products as p', 'p.id', '=', 'i.product_id')
            ->where('s.is_voided', false)
            ->where('p.is_live_bird', true)
            ->when($dari && $sampai, fn ($q) => $q->whereBetween('s.sale_date', [$dari, $sampai]))
            ->selectRaw('coalesce(sum(i.quantity),0) as e, coalesce(sum(i.weight_kg),0) as k')
            ->first();
    }

    /** Catatan ecer pengawas yang belum jadi nota kasir (atau notanya dibatalkan). */
    private function ecerBelumDinota(?string $dari = null, ?string $sampai = null): object
    {
        return DB::table('live_bird_sales as l')
            ->leftJoin('sales as s', 's.id', '=', 'l.sale_id')
            ->where(fn ($q) => $q->whereNull('l.sale_id')->orWhere(fn ($v) => $v->where('s.is_voided', true)->where(fn ($z) => $z->whereNull('l.sumber')->orWhere('l.sumber', '!=', 'KASIR'))))
            ->when($dari && $sampai, fn ($q) => $q->whereBetween('l.sale_date', [$dari, $sampai]))
            ->selectRaw('coalesce(sum(l.birds),0) as e, coalesce(sum(l.weight_kg),0) as k')
            ->first();
    }

    /** Stok ayam hidup saat ini: ['ekor' => int, 'kg' => float]. */
    public function ayamHidup(): array
    {
        $beli = DB::table('purchase_transactions')
            ->selectRaw('coalesce(sum(total_birds),0) as e, coalesce(sum(total_weight_kg),0) as k')->first();
        $mati = DB::table('live_bird_mortalities')
            ->selectRaw('coalesce(sum(birds),0) as e, coalesce(sum(weight_kg),0) as k')->first();
        $prod = DB::table('production_batches')
            ->selectRaw('coalesce(sum(input_birds),0) as e, coalesce(sum(input_weight_kg),0) as k')->first();
        $opn = DB::table('live_bird_opnames')
            ->selectRaw('coalesce(sum(diff_birds),0) as e, coalesce(sum(diff_kg),0) as k')->first();
        $jual = $this->jualEkor();
        $ecer = $this->ecerBelumDinota();

        return [
            'ekor' => (int) round($beli->e - $mati->e - $prod->e - $jual->e - $ecer->e + $opn->e),
            'kg'   => round((float) ($beli->k - $mati->k - $prod->k - $jual->k - $ecer->k + $opn->k), 2),
        ];
    }

    /** Pergerakan ayam hidup pada rentang tanggal. */
    public function arus(Carbon $from, Carbon $to): array
    {
        $f = $from->toDateString();
        $t = $to->toDateString();

        $mati = DB::table('live_bird_mortalities')->whereBetween('record_date', [$f, $t])
            ->selectRaw('coalesce(sum(birds),0) as e, coalesce(sum(weight_kg),0) as k')->first();
        $prod = DB::table('production_batches')->whereBetween('production_date', [$f, $t])
            ->selectRaw('coalesce(sum(input_birds),0) as e, coalesce(sum(input_weight_kg),0) as k')->first();
        $jual = $this->jualEkor($f, $t);
        $ecer = $this->ecerBelumDinota($f, $t);

        return [
            'mati'     => ['e' => (int) $mati->e, 'k' => (float) $mati->k],
            'produksi' => ['e' => (int) $prod->e, 'k' => (float) $prod->k],
            'jual'     => ['e' => (int) ($jual->e + $ecer->e), 'k' => (float) ($jual->k + $ecer->k)],
        ];
    }

    public function opnameAyamTerakhir(int $n = 3): Collection
    {
        return DB::table('live_bird_opnames')
            ->orderByDesc('opname_date')->orderByDesc('created_at')->limit($n)->get();
    }

    /**
     * Status umur simpan: 'lewat' bila umur melebihi batas, 'dekat' pada 1 hari sebelum
     * batas sampai tepat di batas, selainnya 'aman'.
     */
    public function statusUmur(int $umur, int $batas): string
    {
        if ($umur > $batas) {
            return 'lewat';
        }

        return $umur >= max(1, $batas - 1) ? 'dekat' : 'aman';
    }

    /**
     * Stok hasil produksi per item (semua produk non-ayam-hidup, dalam kg).
     * 'hasil' = hasil produksi pada rentang tanggal; 'stok' = stok saat ini.
     *
     * Umur simpan dihitung FIFO: stok yang tersisa dianggap berasal dari hasil produksi
     * terbaru (yang lama sudah keluar duluan). 'lot' = rincian sisa stok per freezer dan
     * tanggal produksi, dari yang paling lama.
     */
    public function produk(Carbon $from, Carbon $to): Collection
    {
        $today = Carbon::today();

        $hasilSemua = DB::table('weighing_records')
            ->selectRaw('product_id, sum(weight_kg) as kg')->groupBy('product_id')->pluck('kg', 'product_id');

        $hasilPeriode = DB::table('weighing_records as w')
            ->join('production_batches as b', 'b.id', '=', 'w.batch_id')
            ->whereBetween('b.production_date', [$from->toDateString(), $to->toDateString()])
            ->selectRaw('w.product_id, sum(w.weight_kg) as kg')->groupBy('w.product_id')->pluck('kg', 'product_id');

        $jual = DB::table('sale_items as i')
            ->join('sales as s', 's.id', '=', 'i.sale_id')
            ->where('s.is_voided', false)
            ->selectRaw('i.product_id, sum(i.quantity) as qty')->groupBy('i.product_id')->pluck('qty', 'product_id');

        $adj = DB::table('stock_adjustments')
            ->selectRaw('product_id, sum(difference_qty) as d')->groupBy('product_id')->pluck('d', 'product_id');

        $dapur = DB::table('kitchen_order_lines as l')
            ->join('kitchen_orders as o', 'o.id', '=', 'l.kitchen_order_id')
            ->where('o.status', '!=', 'VOIDED')
            ->selectRaw('l.product_id, sum(l.quantity_kg) as qty')->groupBy('l.product_id')->pluck('qty', 'product_id');

        $opname = DB::table('stock_adjustments')
            ->orderByDesc('adjustment_date')->orderByDesc('created_at')->get()
            ->unique('product_id')->keyBy('product_id');

        // Semua hasil timbang, terbaru dulu, dikelompokkan per produk.
        $lotRows = DB::table('weighing_records as w')
            ->leftJoin('production_batches as b', 'b.id', '=', 'w.batch_id')
            ->leftJoin('storage_locations as f', 'f.id', '=', 'w.storage_location_id')
            ->select('w.product_id', 'w.weight_kg', 'w.created_at', 'b.production_date', 'f.id as freezer_id', 'f.name as freezer_name')
            ->get()
            ->map(function ($r) {
                $r->tgl = $r->production_date
                    ? Carbon::parse(substr((string) $r->production_date, 0, 10))->startOfDay()
                    : Carbon::parse($r->created_at)->timezone(config('app.timezone'))->startOfDay();

                return $r;
            })
            ->sort(fn ($a, $b) => [$b->tgl->timestamp, (string) $b->created_at] <=> [$a->tgl->timestamp, (string) $a->created_at])
            ->groupBy('product_id');

        return Product::where('is_active', true)->where('is_live_bird', false)
            ->orderBy('name')->get(['id', 'code', 'name', 'max_storage_days'])
            ->map(function ($p) use ($hasilSemua, $hasilPeriode, $jual, $adj, $dapur, $opname, $lotRows, $today) {
                $stok = round((float) (($hasilSemua[$p->id] ?? 0) + ($adj[$p->id] ?? 0) - ($jual[$p->id] ?? 0) - ($dapur[$p->id] ?? 0)), 2);
                $batas = max(1, (int) ($p->max_storage_days ?: 3));

                $sisa = max(0.0, $stok);
                $lot = [];

                foreach ($lotRows->get($p->id, collect()) as $r) {
                    if ($sisa <= 0.004) {
                        break;
                    }

                    $kg = min($sisa, (float) $r->weight_kg);
                    $sisa = round($sisa - $kg, 2);
                    $umur = $r->tgl->lt($today) ? (int) abs($r->tgl->diffInDays($today)) : 0;
                    $kunci = ($r->freezer_id ?? '-') . '|' . $r->tgl->toDateString();

                    if (isset($lot[$kunci])) {
                        $lot[$kunci]->kg = round($lot[$kunci]->kg + $kg, 2);
                        continue;
                    }

                    $lot[$kunci] = (object) [
                        'produk'     => $p->name,
                        'batas'      => $batas,
                        'freezer_id' => $r->freezer_id,
                        'freezer'    => $r->freezer_name ?: 'Belum diatur',
                        'tgl'        => $r->tgl,
                        'kg'         => round($kg, 2),
                        'umur'       => $umur,
                        'status'     => $this->statusUmur($umur, $batas),
                    ];
                }

                $lot = collect(array_values($lot))->sortByDesc('umur')->values();

                $status = $lot->isEmpty()
                    ? 'kosong'
                    : ($lot->contains('status', 'lewat') ? 'lewat' : ($lot->contains('status', 'dekat') ? 'dekat' : 'aman'));

                return (object) [
                    'id'      => $p->id,
                    'name'    => $p->name,
                    'hasil'   => (float) ($hasilPeriode[$p->id] ?? 0),
                    'stok'    => $stok,
                    'opname'  => $opname[$p->id] ?? null,
                    'batas'   => $batas,
                    'lot'     => $lot,
                    'umur'    => $lot->isEmpty() ? null : (int) $lot->max('umur'),
                    'status'  => $status,
                    'freezer' => $lot->pluck('freezer')->unique()->implode(', ') ?: '-',
                ];
            });
    }
}

    {

            'name'      => 'Uji ' . $role,

        ]);

    /** Sisip baris ke tabel; kolom yang tidak ada di tabel dibuang otomatis. */
    private function sisip(string $tabel, array $row): string
    {
        $row['id'] ??= (string) Str::uuid();
        $row += ['created_at' => now(), 'updated_at' => now()];
        $row = array_filter($row, fn ($k) => Schema::hasColumn($tabel, $k), ARRAY_FILTER_USE_KEY);
        DB::table($tabel)->insert($row);

        return $row['id'];
    }

    private function produk(bool $hidup = false): Product
    {
        return Product::create([
            'code'         => 'TST' . Str::upper(Str::random(8)),
            'name'         => ($hidup ? 'Ayam Hidup Uji ' : 'Produk Uji ') . Str::random(4),
            'primary_unit' => $hidup ? 'EKOR' : 'KG',
            'is_active'    => true,
            'is_live_bird' => $hidup,
        ]);
    }

    private function tipePelanggan(): string
    {
        return DB::table('customer_types')->where('is_active', true)->value('id')
            ?: $this->sisip('customer_types', ['code' => 'TST_UJI', 'name' => 'Uji', 'is_active' => true]);
    }

    private function harga(Product $p, string $tipe, float $harga): void
    {
        $this->sisip('product_prices', [
            'product_id' => $p->id, 'customer_type_id' => $tipe, 'price' => $harga, 'is_active' => true,
        ]);
    }

    private function stok(): array
    {
        return app(StokService::class)->ayamHidup();
    }

    private function beli(User $u, int $ekor, float $kg): void
    {
        $supplier = $this->sisip('suppliers', [
            'code' => 'TST' . Str::upper(Str::random(6)), 'name' => 'Pemasok Uji', 'is_active' => true,
        ]);

        $this->actingAs($u)->post('/pembelian', [
            'supplier_id'       => $supplier,
            'purchase_date'     => now()->toDateString(),
            'total_birds'       => $ekor,
            'total_weight_kg'   => $kg,
            'unit_price_per_kg' => 20000,
        ])->assertSessionHasNoErrors()->assertRedirect();
    }

    private function cek(?User $u, string $metode, string $url, int $harap, array $data = []): void
    {
        $req = $u ? $this->actingAs($u) : $this;
        $res = $metode === 'GET' ? $req->get($url) : $req->post($url, $data);

        $this->assertSame($harap, $res->getStatusCode(), ($u->role ?? 'tamu') . " $metode $url");
    }

    public function test_alur_kandang_produksi_opname_dan_angka_dashboard(): void
    {
        $pengawas = $this->user('PENGAWAS_KANDANG');
        $kasir = $this->user('KASIR');
        $admin = $this->user('ADMIN');
        $tgl = now()->toDateString();
        $svc = app(StokService::class);

        $awal = $this->stok();
        $arusAwal = $svc->arus(now()->startOfDay(), now()->startOfDay());

        // Pembelian: stok bertambah
        $this->beli($pengawas, 100, 200);
        $s = $this->stok();
        $this->assertSame($awal['ekor'] + 100, $s['ekor']);
        $this->assertEqualsWithDelta($awal['kg'] + 200, $s['kg'], 0.01);

        // Ayam mati
        $this->actingAs($pengawas)->post('/mortalitas', [
            'record_date' => $tgl, 'birds' => 5, 'weight_kg' => 10, 'cause' => 'Sakit',
        ])->assertSessionHasNoErrors();
        $this->assertSame($awal['ekor'] + 95, $this->stok()['ekor']);

        // Mati melebihi stok -> ditolak
        $this->actingAs($pengawas)->post('/mortalitas', [
            'record_date' => $tgl, 'birds' => $this->stok()['ekor'] + 1, 'weight_kg' => 1,
        ])->assertSessionHasErrors('birds');

        // Produksi: hasil melebihi berat masuk -> ditolak
        $p1 = $this->produk();
        $p2 = $this->produk();
        $this->actingAs($pengawas)->post('/produksi', [
            'production_date' => $tgl, 'input_birds' => 40, 'input_weight_kg' => 90,
            'items' => [['product_id' => $p1->id, 'weight_kg' => 70], ['product_id' => $p2->id, 'weight_kg' => 30]],
        ])->assertSessionHasErrors('items');

        // Produksi: ayam melebihi stok -> ditolak
        $this->actingAs($pengawas)->post('/produksi', [
            'production_date' => $tgl, 'input_birds' => $this->stok()['ekor'] + 1, 'input_weight_kg' => 50,
            'items' => [['product_id' => $p1->id, 'weight_kg' => 10]],
        ])->assertSessionHasErrors('input_birds');

        // Produksi: produk ayam hidup tidak boleh jadi hasil potong
        $hidup = $this->produk(true);
        $this->actingAs($pengawas)->post('/produksi', [
            'production_date' => $tgl, 'input_birds' => 1, 'input_weight_kg' => 5,
            'items' => [['product_id' => $hidup->id, 'weight_kg' => 1]],
        ])->assertSessionHasErrors('items');

        // Produksi valid
        $this->actingAs($pengawas)->post('/produksi', [
            'production_date' => $tgl, 'input_birds' => 40, 'input_weight_kg' => 90,
            'items' => [['product_id' => $p1->id, 'weight_kg' => 60], ['product_id' => $p2->id, 'weight_kg' => 20, 'pieces' => '']],
        ])->assertSessionHasNoErrors();

        $this->assertSame($awal['ekor'] + 55, $this->stok()['ekor']);
        $this->assertEqualsWithDelta(60.0, (float) DB::table('weighing_records')->where('product_id', $p1->id)->sum('weight_kg'), 0.001);
        $this->assertEqualsWithDelta(60.0, (float) $svc->produk(now(), now())->firstWhere('id', $p1->id)->stok, 0.001);

        // Opname ayam hidup: selisih tanpa alasan -> ditolak; dengan alasan -> disimpan.
        // Isian 'system_birds' dari browser harus diabaikan (dihitung server).
        $sis = $this->stok();
        $payload = ['opname_date' => $tgl, 'actual_birds' => $sis['ekor'] - 3, 'actual_kg' => $sis['kg'] - 6, 'system_birds' => 999999];
        $this->actingAs($pengawas)->post('/opname-ayam', $payload)->assertSessionHasErrors('reason');
        $this->actingAs($pengawas)->post('/opname-ayam', $payload + ['reason' => 'Hitung ulang kandang'])->assertSessionHasNoErrors();

        $op = DB::table('live_bird_opnames')->orderByDesc('created_at')->first();
        $this->assertSame($sis['ekor'], (int) $op->system_birds, 'system_birds harus dari server');
        $this->assertSame(-3, (int) $op->diff_birds);
        $this->assertSame($awal['ekor'] + 52, $this->stok()['ekor']);

        // Opname hasil produksi oleh KASIR
        $this->actingAs($kasir)->post('/opname-produk', [
            'adjustment_date' => $tgl, 'fisik' => [$p1->id => 58],
        ])->assertSessionHasErrors('alasan');

        $this->actingAs($kasir)->post('/opname-produk', [
            'adjustment_date' => $tgl, 'fisik' => [$p1->id => 58], 'alasan' => [$p1->id => 'Susut timbang'],
            'system_qty' => 12345,
        ])->assertSessionHasNoErrors();

        $adj = StockAdjustment::where('product_id', $p1->id)->firstOrFail();
        $this->assertEqualsWithDelta(60.0, (float) $adj->system_qty, 0.001);
        $this->assertEqualsWithDelta(-2.0, (float) $adj->difference_qty, 0.001);
        $this->assertEqualsWithDelta(58.0, (float) $svc->produk(now(), now())->firstWhere('id', $p1->id)->stok, 0.001);

        // Id produk palsu -> ditolak
        $this->actingAs($kasir)->post('/opname-produk', [
            'adjustment_date' => $tgl, 'fisik' => [(string) Str::uuid() => 5],
        ])->assertSessionHasErrors('fisik');

        // Angka arus yang dipakai dashboard
        $arus = $svc->arus(now()->startOfDay(), now()->startOfDay());
        $this->assertSame($arusAwal['mati']['e'] + 5, $arus['mati']['e']);
        $this->assertSame($arusAwal['produksi']['e'] + 40, $arus['produksi']['e']);
        $this->assertEqualsWithDelta($arusAwal['produksi']['k'] + 90, $arus['produksi']['k'], 0.01);

        // Dashboard tampil di semua peran, termasuk periode yang tidak valid
        foreach ([$admin, $pengawas, $kasir, $this->user('MANAGER')] as $u) {

        }
    }

    // ---------- 4. POS ayam hidup per ekor + void ----------

    public function test_pos_ayam_hidup_per_ekor_stok_dan_void(): void
    {
        $pengawas = $this->user('PENGAWAS_KANDANG');
        $kasir = $this->user('KASIR');
        $admin = $this->user('ADMIN');

        $this->beli($pengawas, 50, 100);

        $hidup = $this->produk(true);
        $tipe = $this->tipePelanggan();
        $this->harga($hidup, $tipe, 50000);
        $awal = $this->stok();
        $dasar = ['customer_type_id' => $tipe, 'payment_method' => 'CASH'];

        // Tanpa berat -> ditolak
        $this->actingAs($kasir)->post('/kasir', $dasar + ['amount_paid' => 150000,
            'items' => [['product_id' => $hidup->id, 'quantity' => 3]]])->assertSessionHasErrors('items');

        // Ekor pecahan -> ditolak
        $this->actingAs($kasir)->post('/kasir', $dasar + ['amount_paid' => 125000,
            'items' => [['product_id' => $hidup->id, 'quantity' => 2.5, 'weight_kg' => 5]]])->assertSessionHasErrors('items');

        // Melebihi stok -> ditolak
        $n = $awal['ekor'] + 1;
        $this->actingAs($kasir)->post('/kasir', $dasar + ['amount_paid' => $n * 50000,
            'items' => [['product_id' => $hidup->id, 'quantity' => $n, 'weight_kg' => 10]]])->assertSessionHasErrors('items');

        $this->assertSame($awal['ekor'], $this->stok()['ekor'], 'Transaksi gagal tidak boleh mengubah stok');

        // Valid (dengan percobaan menitipkan harga dari browser, harus diabaikan)
        $this->actingAs($kasir)->post('/kasir', $dasar + ['amount_paid' => 150000,
            'items' => [['product_id' => $hidup->id, 'quantity' => 3, 'weight_kg' => 6.5, 'unit_price' => 1]]])
            ->assertSessionHasNoErrors();

        $sale = Sale::where('cashier_id', $kasir->id)->latest('created_at')->firstOrFail();
        $this->assertEqualsWithDelta(150000.0, (float) $sale->total_amount, 0.001);
        $this->assertEqualsWithDelta(6.5, (float) DB::table('sale_items')->where('sale_id', $sale->id)->value('weight_kg'), 0.001);
        $this->assertSame($awal['ekor'] - 3, $this->stok()['ekor']);
        $this->assertEqualsWithDelta($awal['kg'] - 6.5, $this->stok()['kg'], 0.01);

        // Void: kasir ditolak, admin berhasil, stok kembali, void ganda ditolak
        $this->actingAs($kasir)->post("/kasir/{$sale->id}/void", ['void_reason' => 'uji batal'])->assertForbidden();
        $this->actingAs($admin)->post("/kasir/{$sale->id}/void", ['void_reason' => 'uji batal'])->assertSessionHasNoErrors();
        $this->assertTrue((bool) $sale->refresh()->is_voided);
        $this->assertSame($awal['ekor'], $this->stok()['ekor'], 'Void harus mengembalikan stok ayam hidup');
        $this->actingAs($admin)->post("/kasir/{$sale->id}/void", ['void_reason' => 'uji batal'])->assertSessionHasErrors('void_reason');
    }
